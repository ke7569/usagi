cmake_minimum_required(VERSION 3.10)
project(usagi CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SZE_MARCH_NATIVE "Compile for the build machine CPU" OFF)
option(USAGI_BUILD_DEEPWIN "Build the legacy Deepwin strategy and MD plugins" OFF)
option(SZE_BUILD_TD "Build the Shenzhen ATP TD plugin" OFF)
option(SSE_BUILD_TD "Build the Shanghai ATP TD plugin" OFF)
option(T0_BUILD_SZE_STREAM_PROCESSOR "Build the standalone Shenzhen stream processor" ON)
option(T0_BUILD_SSE_STREAM_PROCESSOR "Build the standalone Shanghai stream processor" ON)
option(SZE_BUILD_DIRECT_SHARDED_RING "Build experimental Shenzhen sharded-ring tests/benchmark" OFF)

set(USAGI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(SZE_TD_API_DIR "${USAGI_ROOT}/adapters/td/atp/api" CACHE PATH
    "ATP Quant API root containing include/ and lib/")
set(RUNTIME "${USAGI_ROOT}/runtime_so" CACHE PATH
    "Deepwin runtime root; supply explicitly for SDK builds")
set(DEEPWIN_INCLUDE "${USAGI_ROOT}/toolchain/deepwin_include")
set(BOOST_INCLUDE "${USAGI_ROOT}/toolchain/boost_include")
set(PYTHON_INCLUDE "${USAGI_ROOT}/toolchain/python_include")
set(WC_LIB "${RUNTIME}/deepwin_core/lib/wingchun")
set(YJJ_LIB "${RUNTIME}/deepwin_core/lib/yijinjing")
set(TPWC_LIB "${RUNTIME}/third_party/wingchun")
set(BOOST_LIB "${RUNTIME}/third_party/boost")

if(SZE_MARCH_NATIVE)
  add_compile_options(-march=native)
else()
  add_compile_options(-march=x86-64 -mtune=generic)
endif()
add_compile_options(-O3 -DNDEBUG)
# Preserve existing external binary/library locations and exported names.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include_directories("${USAGI_ROOT}")
enable_testing()
find_program(T0_STREAM_PYTHON NAMES python3 python3.6)

add_subdirectory(common)
add_subdirectory(sze)
add_subdirectory(sse)
add_subdirectory(apps)
add_subdirectory(tests)
if(USAGI_BUILD_DEEPWIN OR SZE_BUILD_TD OR SSE_BUILD_TD)
  add_subdirectory(adapters/deepwin)
endif()
